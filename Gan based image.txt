import torch
import torch.nn as nn
import torch.nn.functional as F
import matplotlib.pyplot as plt
from torchvision import datasets

x = datasets.MNIST('.', download=True).data[:5008].float().div(255).unsqueeze(1)
t = F.max_pool2d(x, 3, 1, 1)

G = nn.Sequential(
    nn.Conv2d(1, 16, 3, padding=1),
    nn.ReLU(),
    nn.Conv2d(16, 16, 3, padding=1),
    nn.ReLU(),
    nn.Conv2d(16, 1, 3, padding=1),
    nn.Sigmoid()
)

D = nn.Sequential(
    nn.Conv2d(2, 16, 4, 2, 1),
    nn.LeakyReLU(0.2),
    nn.Conv2d(16, 32, 4, 2, 1),
    nn.LeakyReLU(0.2),
    nn.Flatten(),
    nn.Linear(32 * 7 * 7, 1)
)

bce = nn.BCEWithLogitsLoss()
oG = torch.optim.Adam(G.parameters(), 1e-3, betas=(0.5, 0.999))
oD = torch.optim.Adam(D.parameters(), 1e-3, betas=(0.5, 0.999))
ones, zeros = torch.ones(50, 1), torch.zeros(50, 1)

for e in range(5):
    for i in range(0, 5000, 50):
        s, r = x[i:i+50], t[i:i+50]
        f = G(s)

        ld = bce(D(torch.cat([s, r], 1)), ones) + bce(
            D(torch.cat([s, f.detach()], 1)), zeros
        )
        oD.zero_grad()
        ld.backward()
        oD.step()

        lg = bce(D(torch.cat([s, f], 1)), ones) + 100 * F.l1_loss(f, r)
        oG.zero_grad()
        lg.backward()
        oG.step()

    print('epoch', e+1, 'D loss', round(ld.item(), 3), 'G loss', round(lg.item(), 3))

s, r = x[5000:], t[5000:]
with torch.no_grad():
    f = G(s)

print(
    'L1 original vs target:', round(F.l1_loss(s, r).item(), 4),
    '| GAN output vs target:', round(F.l1_loss(f, r).item(), 4)
)

rows = [s, r, f]
for k in range(3):
    for j in range(8):
        plt.subplot(3, 8, k*8 + j + 1)
        plt.imshow(rows[k][j, 0], cmap='gray')
        plt.axis('off')

plt.suptitle('Row1: Original | Row2: Target style | Row3: GAN output')
plt.show()
