import torch
import torch.nn.functional as F

N, D = 2000, 10
x = torch.zeros(N, D)
x[:, 0] = torch.bernoulli(torch.full((N,), 0.5))

for i in range(1, D):
    x[:, i] = (x[:, i-1] + (torch.rand(N) < 0.1).float()) % 2

tr, te = x[:1600], x[1600:]

W = torch.zeros(D, D, requires_grad=True)
b = torch.zeros(D, requires_grad=True)
mask = torch.tril(torch.ones(D, D), -1)

f = lambda x: x @ (W * mask).T + b
opt = torch.optim.Adam([W, b], 0.05)

for e in range(300):
    loss = F.binary_cross_entropy_with_logits(f(tr), tr)
    opt.zero_grad()
    loss.backward()
    opt.step()
    if e % 50 == 0:
        print('epoch', e, 'train loss', round(loss.item(), 4))

with torch.no_grad():
    l = f(te)
    print('Test loss:', round(F.binary_cross_entropy_with_logits(l, te).item(), 4))
    print('Test accuracy:', round((((l > 0).float() == te).float().mean()).item(), 4))

    s = torch.zeros(10, D)
    for i in range(D):
        s[:, i] = torch.bernoulli(torch.sigmoid(f(s)[:, i]))

print(s.int())
